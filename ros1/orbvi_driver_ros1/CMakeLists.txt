cmake_minimum_required(VERSION 3.16)
project(orbvi_driver_ros1)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs std_msgs diagnostic_msgs message_generation)
find_package(orbvi_driver REQUIRED)
add_message_files(FILES FrameMetadata.msg ImuMetadata.msg)
generate_messages(DEPENDENCIES std_msgs)
catkin_package(CATKIN_DEPENDS roscpp sensor_msgs std_msgs diagnostic_msgs message_runtime)
add_executable(orbvi_node ../../wrappers/node.cpp)
target_include_directories(orbvi_node PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(orbvi_node orbvi_driver::driver ${catkin_LIBRARIES})
add_dependencies(orbvi_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
install(TARGETS orbvi_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
add_executable(orbvi_ros_check ../../wrappers/check.cpp)
target_include_directories(orbvi_ros_check PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(orbvi_ros_check orbvi_driver::config ${catkin_LIBRARIES})
add_dependencies(orbvi_ros_check ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
install(TARGETS orbvi_ros_check RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

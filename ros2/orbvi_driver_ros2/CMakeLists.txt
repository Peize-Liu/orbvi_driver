cmake_minimum_required(VERSION 3.16)
project(orbvi_driver_ros2)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(orbvi_driver REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/FrameMetadata.msg" "msg/ImuMetadata.msg" DEPENDENCIES std_msgs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(orbvi_node ../../wrappers/node.cpp)
target_compile_definitions(orbvi_node PRIVATE ORBVI_ROS2)
target_link_libraries(orbvi_node orbvi_driver::driver ${typesupport})
ament_target_dependencies(orbvi_node rclcpp sensor_msgs diagnostic_msgs)
install(TARGETS orbvi_node DESTINATION lib/${PROJECT_NAME})
add_executable(orbvi_ros_check ../../wrappers/check.cpp)
target_compile_definitions(orbvi_ros_check PRIVATE ORBVI_ROS2)
target_link_libraries(orbvi_ros_check orbvi_driver::config ${typesupport})
ament_target_dependencies(orbvi_ros_check rclcpp sensor_msgs)
install(TARGETS orbvi_ros_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()

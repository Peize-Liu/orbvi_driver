cmake_minimum_required(VERSION 3.16)
project(orbvi_driver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(ORBVI_BUILD_RECEIVER "Build C++ data receiver (FFmpeg required)" ON)
option(ORBVI_BUILD_TESTS "Build driver tests" ON)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(orbvi_config STATIC src/config.cpp src/board.cpp)
set_target_properties(orbvi_config PROPERTIES POSITION_INDEPENDENT_CODE ON EXPORT_NAME config)
target_include_directories(orbvi_config PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(orbvi_config PUBLIC yaml-cpp)
add_executable(orbvi_board src/board_main.cpp)
target_link_libraries(orbvi_board PRIVATE orbvi_config Threads::Threads)
install(TARGETS orbvi_config EXPORT orbvi_driverTargets ARCHIVE DESTINATION lib)
install(TARGETS orbvi_board RUNTIME DESTINATION bin)
if(ORBVI_BUILD_RECEIVER)
  set(ORBVI_SDK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/orbvi-4p-sdk" CACHE PATH "SDK source; empty selects installed package")
  if(ORBVI_SDK_SOURCE_DIR)
    set(ORBVI_BUILD_TESTS_SAVED ${ORBVI_BUILD_TESTS})
    set(ORBVI_BUILD_TESTS OFF CACHE BOOL "SDK tests" FORCE)
    set(ORBVI_BUILD_EXAMPLES OFF CACHE BOOL "SDK examples" FORCE)
    add_subdirectory(${ORBVI_SDK_SOURCE_DIR} sdk)
    set(ORBVI_BUILD_TESTS ${ORBVI_BUILD_TESTS_SAVED} CACHE BOOL "Driver tests" FORCE)
  else()
    find_package(orbvi_sdk REQUIRED)
  endif()
  add_library(orbvi_driver SHARED src/driver.cpp)
  add_library(orbvi_driver::driver ALIAS orbvi_driver)
  set_target_properties(orbvi_driver PROPERTIES EXPORT_NAME driver INSTALL_RPATH "$ORIGIN")
  target_include_directories(orbvi_driver PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(orbvi_driver PUBLIC orbvi_config Threads::Threads PRIVATE orbvi_sdk::shared)
  add_executable(orbvi_capture src/capture_main.cpp)
  set_target_properties(orbvi_capture PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  target_link_libraries(orbvi_capture PRIVATE orbvi_driver)
  install(TARGETS orbvi_driver EXPORT orbvi_driverTargets LIBRARY DESTINATION lib)
  install(TARGETS orbvi_capture RUNTIME DESTINATION bin)
endif()
if(ORBVI_BUILD_TESTS)
  enable_testing()
  add_executable(test_config tests/test_config.cpp)
  target_link_libraries(test_config PRIVATE orbvi_config)
  add_test(NAME config_and_capabilities COMMAND test_config ${CMAKE_CURRENT_SOURCE_DIR}/config/orbvi.yaml)
endif()
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config/ DESTINATION share/orbvi_driver/config)
install(EXPORT orbvi_driverTargets NAMESPACE orbvi_driver:: DESTINATION lib/cmake/orbvi_driver)
configure_package_config_file(cmake/orbvi_driverConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/orbvi_driverConfig.cmake INSTALL_DESTINATION lib/cmake/orbvi_driver)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/orbvi_driverConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/orbvi_driverConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/orbvi_driverConfigVersion.cmake DESTINATION lib/cmake/orbvi_driver)
